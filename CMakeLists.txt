option(DNS_BENCHMARKS "Build isolated profiling/benchmark executables" OFF)

    src/DNSName.cpp
    src/DNSHeaderParser.cpp
    src/DNSNameDecoder.cpp
    src/DNSRecordParser.cpp
    src/DNSMessageParser.cpp)

function(dns_settings target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(DNS_SANITIZE)
            target_compile_options(${target} PRIVATE /fsanitize=address /Z7)
            target_link_options(${target} PUBLIC /INCREMENTAL:NO /DEBUG)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
        if(DNS_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
